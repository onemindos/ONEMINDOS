# Unsafe operations (POST, PUT, PATCH, DELETE) that do not take an Idempotency-Key header on purpose
# (rule `idempotency`). One line each: METHOD /path  # why. A gateway that replays a stored answer
# for a repeated key must never store a secret, so routes that answer with fresh credentials are
# here; so are routes whose own protocol already makes a retry safe.
POST /v1/auth/login/options  # read only: returns a fresh challenge, nothing changes
POST /v1/auth/login  # answers with fresh tokens, which a replay store must not keep; a retry signs in again
POST /v1/auth/refresh  # answers with fresh tokens; refresh-token rotation already rejects a replay
DELETE /v1/auth/app-session  # clearing the caller's own cookie twice is the same as once
POST /v1/auth/logout  # ending the caller's own session twice is the same as once
POST /v1/auth/approver-keys  # a retry with the same public key answers 409 (already registered), so a repeat never adds a second key (#702)
DELETE /v1/auth/approver-keys/{key_id}  # revoking twice is the same as once: a repeat finds no live key and changes nothing (#702)
POST /v1/sites  # The caller supplies a unique site ID. A retry with that ID returns 409 before writing a site or a second creation audit record.
PUT /v1/me/dashboard-card  # replaces the caller's single card; identical saves leave state and audit unchanged
DELETE /v1/me/devices/{device_id}  # Lost this device (#715) is idempotent by design: a repeat revokes nothing and answers removed: false
PATCH /v1/files/uploads/{upload_id}  # tus: the Upload-Offset header makes a retried chunk safe
PATCH /v1/photos/uploads/{upload_id}  # tus: the Upload-Offset header makes a retried chunk safe
POST /v1/actions  # contract §12.1: keeps its §3 body field idempotency_key instead of the header
POST /v1/tips/{tip_id}/state  # sets one tip's state to the given value: setting the same state twice is the same as once
POST /v1/files/wopi/tokens  # answers with a fresh WOPI access token, which a replay store must not keep; a retry mints another
POST /v1/files/wopi/files/{file_id}  # WOPI (#675): an office engine sends no Idempotency-Key; Lock, RefreshLock and Unlock repeated with the same X-WOPI-Lock give the same result
POST /v1/files/wopi/files/{file_id}/contents  # WOPI PutFile (#675): needs the file's current lock and its last-saved version, so a retried save writes the same bytes again
POST /v1/files/shares  # answers with the link secret once, which a replay store must not keep; a retry makes a second link the creator can revoke
DELETE /v1/files/shares/{share_id}  # revoking a link twice is the same as once: the second answer is the same revoked link
POST /v1/os/rollouts/{rollout_id}/cancel  # cancelling twice leaves the same state: the second answer is 409 wrong_state and changes nothing
PUT /v1/agents/{name}/nats-key  # sets one agent's one key: the same body twice leaves the same state
DELETE /v1/agents/{name}/nats-key  # removing an agent's key twice is the same as once
POST /v1/host-grants/requests  # files a host.grant action and keeps its §3 body field idempotency_key, like POST /v1/actions
POST /v1/host-grants/{grant_id}/revoke  # revoking twice leaves the grant revoked; the repeat answers 404 and changes nothing
POST /v1/bridges/{bridge_id}/events  # every event carries its own id and revision: a retried batch counts as duplicates, never twice
POST /v1/bridges/{bridge_id}/resume  # resuming a bridge that already runs changes nothing
POST /v1/files/app-passwords  # answers with a fresh app password, which a replay store must never keep
POST /v1/files/app-passwords/{password_id}/revoke  # revoking twice is the same as once (the second is 404)
POST /v1/sync/chunks/missing  # read only: asks which chunks the root lacks; nothing changes but a chunk's lease
PUT /v1/sync/chunks/{chunk_hash}  # content-addressed: the path is the bytes' sha256, so a retried chunk is the same chunk (stored: false)
POST /v1/sync/commit  # conditional on base_revision / If-Match: a retry after a commit landed is 412 with the new revision, never a second write
POST /v1/sync/locks  # advisory (#699): a retried take only gives the same holder a second lock that never blocks that holder and expires on its own
POST /v1/sync/locks/{lock_id}/refresh  # sets expiry to now + ttl: a retry lands on the same expiry
DELETE /v1/sync/locks/{lock_id}  # releasing twice changes nothing (the second is 404)
POST /v1/radios/{radio_id}/confirm  # an upsert of one radio's confirmation (#732): confirming twice stores the same row and changes nothing
DELETE /v1/agent-browser/sessions/{session_id}  # ending a session twice is the same as once (the second answers the ended session)
DELETE /v1/agent-browser/sites/{site_id}  # removing an approved site twice leaves the same allowlist (the second answers 404)
PATCH /v1/work/projects/{project_id}/tasks/{task_id}  # guarded by If-Match; replay cannot change a task at a newer revision
DELETE /v1/work/projects/{project_id}/tasks/{task_id}  # the second removal sees no task
POST /v1/custom-apps/plan  # asking again for the same commit refreshes that revision's scan and never adds a second revision; nothing is installed
POST /v1/custom-apps/{app_id}/rollback  # a repeat picks the same earlier revision and points the catalog at it again; the update that applies it is its own approved action
POST /v1/approval-policies/{policy_id}/confirm  # re-enables one paused rule: a repeat finds no paused rule, answers 404 and changes nothing
PUT /v1/agents/{name}/inputs  # replaces the whole declaration: the same body twice leaves the same state
POST /v1/work/parse  # read only: previews quick-add text and creates nothing
PATCH /v1/work/projects/{project_id}  # sets fields to the stated values, so a repeat changes nothing more
DELETE /v1/work/projects/{project_id}  # the second removal sees no project
PUT /v1/work/projects/{project_id}/members/{person_id}  # adding a member twice is the same as once
DELETE /v1/work/projects/{project_id}/members/{person_id}  # removing a member twice is the same as once
DELETE /v1/work/projects/{project_id}/tasks/{task_id}/comments/{comment_id}  # the second removal sees no comment
PATCH /v1/work/projects/{project_id}/sections/{section_id}  # sets the name to the stated value, so a repeat changes nothing more
DELETE /v1/work/projects/{project_id}/sections/{section_id}  # the second removal sees no section
POST /v1/auth/recovery/contacts  # answers with a fresh signed invite whose nonce a replay store must not keep; a retry makes a second invite the owner can withdraw (the 5-contact limit still holds)
DELETE /v1/auth/recovery/contacts/{contact_id}  # removing twice is the same as once: the repeat answers the removed contact and changes nothing
PUT /v1/auth/recovery/settings  # sets each named wait to the given option: the same body twice leaves the same state
POST /v1/recovery/invites/{invite_id}/accept  # answers with the contact token and secret once, which a replay store must not keep; the single-use invite refuses a retry (409 invite_used)
POST /v1/recovery/contact/resign  # resigning twice is the same as once: the token stops at the first, so a repeat is refused before it changes anything
